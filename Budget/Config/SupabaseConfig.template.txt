import Foundation

/// Supabase Configuration Template
///
/// Instructions:
/// 1. Copy this file and rename it to `SupabaseConfig.swift`
/// 2. Replace the placeholder values with your actual Supabase credentials
/// 3. The actual `SupabaseConfig.swift` is gitignored for security
///
/// To get your credentials:
/// 1. Go to your Supabase project dashboard
/// 2. Click on Settings > API
/// 3. Copy the Project URL and anon/public key
/// 4. For Google OAuth, get your client IDs from Google Cloud Console

enum SupabaseConfig {
    /// Your Supabase project URL (e.g., "https://xxxxx.supabase.co")
    static let url = URL(string: "YOUR_SUPABASE_PROJECT_URL")!

    /// Your Supabase anon/public key
    static let anonKey = "YOUR_SUPABASE_ANON_KEY"

    // Google OAuth Client IDs (optional, only if using Google Sign In)
    /// Web client ID from Google Cloud Console
    static let googleWebClientID = "YOUR_GOOGLE_WEB_CLIENT_ID.apps.googleusercontent.com"

    /// iOS client ID from Google Cloud Console
    static let googleiOSClientID = "YOUR_GOOGLE_IOS_CLIENT_ID.apps.googleusercontent.com"
}
